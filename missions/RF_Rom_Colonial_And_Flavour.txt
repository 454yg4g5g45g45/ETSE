roman_colonial_missions_1 = {
	slot = 1
	generic = no
	ai = yes
	potential = {
		tag = ROM
		NOT = { map_setup = map_setup_random }
	}
	has_country_shield = yes
	
	rom_borealis_terra_mission = {
		icon = rf_mission_rom_settlers_north_america
		required_missions = { rom_mare_novum_mission }
		position = 14
		provinces_to_highlight = {
			OR = {
				colonial_region = colonial_eastern_america
				colonial_region = colonial_canada
				colonial_region = colonial_louisiana
			}
			NOT = { country_or_non_sovereign_subject_holds = ROOT }
			has_discovered = ROOT
		}
		trigger = {
			custom_trigger_tooltip = {
				tooltip = rom_borealis_terra_mission.tt
				any_subject_country = {
					is_colonial_nation = yes
					capital_scope = {
						OR = {
							colonial_region = colonial_eastern_america
							colonial_region = colonial_canada
							colonial_region = colonial_louisiana
						}
					}
				}
			}
		}
		effect = {
			add_dip_power = 75
			add_adm_power = 75
			add_mil_power = 75
			add_prestige = 25
			every_subject_country = {
				limit = {
					is_colonial_nation = yes
					capital_scope = {
						OR = {
							colonial_region = colonial_eastern_america
							colonial_region = colonial_canada
							colonial_region = colonial_louisiana
						}
					}
				}
				add_prestige = 20
				add_liberty_desire = -10
			}
		}
	}
	
	rom_imperium_mexicanum_mission = {
		icon = rf_mission_rom_aztec_temple
		required_missions = { rom_aurum_novum_mission } 
		position = 15 
		provinces_to_highlight = {
			colonial_region = colonial_mexico
			NOT = { country_or_non_sovereign_subject_holds = ROOT }
			has_discovered = ROOT
		}
		trigger = {
			any_subject_country = {
				capital_scope = { colonial_region = colonial_mexico }
				is_colonial_nation = yes
			}
		}
		effect = {
			add_mil_power = 100
			every_subject_country = {
				limit = {
					capital_scope = { colonial_region = colonial_mexico }
				}
				add_army_tradition = 20
				add_liberty_desire = -10
				# Historical flavor: "New Spain" had the first printing press in Americas. Rome would build a library.
				add_country_modifier = {
					name = "rom_library_of_mexico"
					duration = -1
				}
			}
		}
	}

	rom_limes_californianus_mission = {

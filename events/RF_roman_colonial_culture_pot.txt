#Sudan
#country culture
country_event = {
    id = roman_colonial_culture_pot.1
    title = roman_colonial_culture_pot.1.t
    desc = roman_colonial_culture_pot.1.d
    picture = CULTURE_eventPicture

    trigger = {
        is_colonial_nation = yes
        overlord = { tag = ROM }
        capital_scope = { colonial_region = colonial_sudan }
        NOT = { primary_culture = nilassian }
    }

    mean_time_to_happen = {
        months = 24
        modifier = {
            factor = 0.5
            overlord = { has_country_modifier = pax_romana }
        }
    }

    option = {
        name = roman_colonial_culture_pot.1.a
        ai_chance = { factor = 95 }
        change_primary_culture = nilassian
		set_country_flag = african_culture_change_event
		capital_scope = {
            change_culture = nilassian
            add_province_modifier = {
                name = romanized_identity
                duration = 7300  # ~20 years
            }
        }
    }

    option = {
        name = roman_colonial_culture_pot.1.b
        ai_chance = { factor = 5 }
    }
}

#province culture
country_event = {
    id = roman_colonial_culture_pot.2
    title = roman_colonial_culture_pot.2.t
    desc = roman_colonial_culture_pot.2.d
    picture = CULTURE_eventPicture

    trigger = {
        is_colonial_nation = yes
        primary_culture = nilassian
        overlord = { tag = ROM }
        any_owned_province = {
            OR = { 
				colonial_region = colonial_sudan
				area = red_sea_coast_area
				area = lower_nubia_area
			}
            NOT = { culture = nilassian }
            has_owner_culture = no
            has_owner_religion = yes
            NOT = { unrest = 0 }
            NOT = { nationalism = 1 }
        }
    }

    mean_time_to_happen = {
        months = 360
        modifier = {
            factor = 0.5
            has_country_modifier = pax_romana
        }
        modifier = {
            factor = 0.8
            stability = 3
        }
        modifier = {
            factor = 1.25
            NOT = { stability = 1 }
        }
    }

    option = {
        name = roman_colonial_culture_pot.2.a
        ai_chance = { factor = 90 }

        random_owned_province = {
            limit = {
				colonial_region = colonial_sudan
				NOT = { culture = nilassian }
				has_owner_culture = no
				has_owner_religion = yes
				NOT = { unrest = 0 }
				NOT = { nationalism = 1 }
            }
            change_culture = nilassian
            add_province_modifier = {
                name = romanized_identity
                duration = 7300    # ~20 years
            }
        }
    }

    option = {
        name = roman_colonial_culture_pot.2.b
        ai_chance = { factor = 10 }
        add_prestige = 2
    }
}

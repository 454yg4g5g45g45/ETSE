# --------------------------------------------------
# EVENT 1: POSITIVE (Powerful) - The Living Apotheosis
# Context: The Collegium declares you a God while still alive.
# --------------------------------------------------
country_event = {
    id = imperial_cult_flavor.1
    title = "imperial_cult_flavor.1.t"
    desc = "imperial_cult_flavor.1.d"
    picture = RELIGION_eventPicture
    
    trigger = {
        religion = zz_imperial_cult
        NOT = { has_ruler_modifier = imperial_apotheosis }
        OR = {
			adm = 6
			dip = 6
			mil = 6
		}
    }
    
    mean_time_to_happen = { months = 480 }
    
    option = {
        name = "imperial_cult_flavor.1.a" # Numen Augusti!
        add_stability = 1
        add_patriarch_authority = 25
        add_prestige = 50
		add_republican_tradition = 10
		add_legitimacy = 15
		add_devotion = 15
        
        add_ruler_modifier = {
            name = imperial_apotheosis
            duration = -1
        }
    }
}

# Disaster start
country_event = {
	id = great_plague.1
	title = "great_plague.1.t"
	desc = "great_plague.1.d"
	picture = PLAGUE_eventPicture
		
	is_triggered_only = yes
	
	immediate = {
		set_country_flag= has_the_great_plague
		random_owned_province = {
			set_country_flag = had_the_great_plague
			limit = { 
				development = 12
				NOT = { has_province_modifier = great_plague_province } 
			}
			add_devastation = 25
			if = {
				limit = { base_tax = 2 NOT = { base_tax = 4 } }
				add_base_tax = -1
			}
			if = {
				limit = { base_tax = 4 NOT = { base_tax = 7 } }
				add_base_tax = -2
			}
			if = {
				limit = { base_tax = 8 NOT = { base_tax = 11 } }
				add_base_tax = -3
			}
			if = {
				limit = { base_tax = 12 }
				add_base_tax = -4
			}
			if = {
				limit = { base_production = 2 NOT = { base_production = 4 } }
				add_base_production = -1
			}
			if = {
				limit = { base_production = 4 NOT = { base_production = 7 } }
				add_base_production = -2
			}
			if = {
				limit = { base_production = 8 NOT = { base_production = 11 } }
				add_base_production = -3
			}
			if = {
				limit = { base_production = 12 }
				add_base_production = -4
			}
			if = {
				limit = { base_manpower = 2 NOT = { base_manpower = 4 } }
				add_base_manpower = -1
			}
			if = {
				limit = { base_manpower = 4 NOT = { base_manpower = 7 } }
				add_base_manpower = -2
			}
			if = {
				limit = { base_manpower = 8 NOT = { base_manpower = 11 } }
				add_base_manpower = -3
			}
			if = {
				limit = { base_manpower = 12 }
				add_base_manpower = -4
			}
			add_province_modifier = {
				name = great_plague_province
				duration = 1825
			}
		}
	}

	option = {
		name = "great_plague.1.a"
		add_stability = -2
	}
}

# Plague Spreads
province_event = {
	id = great_plague.3
	title = "great_plague.3.t"
	desc = "great_plague.3.d"
	picture = PLAGUE_eventPicture
	#hidden=yes
	trigger = {
		owner = { has_disaster = great_plague }
		AND = {
			# NOT = {
			# 	owner = {
			# 		OR = {
			# 			technology_group=chinese
			# 		    technology_group=indian
			# 		}
					
			# 	}
			# }
			OR = {
				province_has_center_of_trade_of_level = 1
				AND = {
					# Previously infected provinces can be reinfected, the chance of it happening is just lower.
					NOT = { has_province_modifier = great_plague_province }
					#NOT = { has_province_flag = had_great_plague }
	
					OR = {
						any_neighbor_province = { has_province_modifier = great_plague_province }
						AND = { 
							has_port = yes
							any_province = { 
								region = ROOT
								has_province_modifier = great_plague_province
							}
						}
					}
				}
				owner = { #Trigger plague spread in random provinces if disaster is active, but not plague provinces exist in nation. Important for island nations like England, which can only transfer the disease through alliances with mainland European nations.
					has_disaster = great_plague
					NOT = {
						any_owned_province = {
							has_province_flag=had_great_plague
						}
					}
			}
			}
			#NOT = {
			#	owner = {
			#		
			#	}
			#}
			
		}
	}

	# delta-n is the number of provinces an infected province can infect before it's no longer infected.
	# If delta-n <1, the disease will die out, else, it grows.
	# 5 years is ~ 60 months
	# Assuming each province has 4 uninfected neighbors (I have not verified this)
	#When mtth = 3 
	# delta-n = 

	#Factors of spread
	#Higher development increases spread
	#Previously infected reduces spread
	#Having medicine reduces spread
	#If dev > 60, infection is instant.
	mean_time_to_happen = {
		months = 48
		modifier = {
			factor = 0.5
			development = 5
		}
		modifier = {
			factor = 0.5
			development = 10
		}
		modifier = {
			factor = 0.5
			development = 20
		}
		modifier = {
			factor = 0.5
			development = 30
		}
		modifier = {
			factor = 0.5
			development = 40
		}
		modifier = {
			factor = 0.5
			development = 50
		}
		modifier = {
			factor = 0.01
			development = 60
		}
		modifier = {
			factor = 8
			has_province_flag = had_great_plague
		}
		modifier = {
			factor = 5
			owner = {
				has_institution= medicine
			}
		}
	}
	immediate = {
		set_province_flag = had_great_plague
		if = {
			limit = {is_emperor_of_china=no}
			add_devastation = 25
		}
		else = {
			add_devastation = 5 #Emperor of China is uniquely vulnerable to high devastation, so to prevent devastation so high that it prevents a new chinese dynasty from rising and surviving, we must apply less devastation to the China region.
		}
		if = {
			limit = {

				owner = {
					OR ={
						is_elector = yes
						is_emperor = yes
					}
				}


			}
			#emperor = {
				#add_imperial_influence=-5
			#}
		}
		if = { #The great Plague hit Europe particularly hard.
			limit = {continent = europe}
			add_random_development = -2
		}
		if = { #The great Plague hit Europe particularly hard.
			limit = {continent = africa}
			add_random_development = -1
		}
		
		#if = {
		#	limit = { base_tax = 2 NOT = { base_tax = 4 } }
		#	add_base_tax = -1
		#}
		if = {
			limit = { base_tax = 4 NOT = { base_tax = 7 } }
			add_base_tax = -1
		}
		if = {
			limit = { base_tax = 8 NOT = { base_tax = 10 } }
			add_base_tax = -2
		}
		if = {
			limit = { base_tax = 11 NOT = { base_tax = 13 } }
			add_base_tax = -3
		}
		if = {
			limit = { base_tax = 14 NOT = { base_tax = 100 } }
			add_base_tax = -4
		}
		#if = {
		#	limit = { base_production = 2 NOT = { base_production = 4 } }
		#	add_base_production = -1
		#}
		if = {
			limit = { base_production = 4 NOT = { base_production = 7 } }
			add_base_production = -1
		}
		if = {
			limit = { base_production = 8 NOT = { base_production = 10 } }
			add_base_production = -2
		}
		if = {
			limit = { base_production = 11 NOT = { base_production = 13 } }
			add_base_production = -3
		}
		if = {
			limit = { base_production = 14 NOT = { base_production = 100 } }
			add_base_production = -4
		}
		#if = {
		#	limit = { base_manpower = 2 NOT = { base_manpower = 4 } }
		#	add_base_manpower = -1
		#}
		if = {
			limit = { base_manpower = 4 NOT = { base_manpower = 7 } }
			add_base_manpower = -1
		}
		if = {
			limit = { base_manpower = 8 NOT = { base_manpower = 10 } }
			add_base_manpower = -2
		}
		if = {
			limit = { base_manpower = 11 NOT = { base_manpower = 13 } }
			add_base_manpower = -3
		}
		if = {
			limit = { base_manpower = 14 NOT = { base_manpower = 100 } }
			add_base_manpower = -4
		}
		
	}
	#After disease strikes province, country can either let the province spread more disease, or quarantine the province to limit the damage to other provinces.
	#If nation has the ability to, they will quarantine the province 5/6 times.
	option = {
		name = "great_plague.3.a"
		ai_chance = { factor = 10
		}
		add_province_modifier = {
			name = great_plague_province
			duration = 1825
		}
	}
	option = {
		name="great_plague.3.b"
		highlight = yes
		ai_chance = {factor = 50}
		trigger = {
			adm_power=50
			OR = {
				
				religion = jewish
				tech_group=indian
				owner = {
					OR = {
						ruler_has_personality= scholar_personality
						has_institution=medicine
					}
					
				}
			}
			
		}
		owner = {
			add_adm_power=-50
		}
	}
}

namespace = mmetrFlavor_War

#Events related to warfare and sieges, including flavor events like for the Qarmatians and Mongols

# Sack of Basra - Qarmatians
province_event = {
	id = mmetrFlavor_War.1
	title = "mmetrFlavor_War.1.t"
	desc = "mmetrFlavor_War.1.desc"
	picture = SIEGE_eventPicture
	
	trigger = {
		province_id = 408		#Basra
		controlled_by = QRM
		NOT = {
			owned_by = QRM
		}
		owner = {
			OR = {
				tag = ABA
				was_tag = ABA
				has_reform = caliphate_reform
				has_reform = caliphate_theocratic_reform
				tag = SEL		#another major power that existed alongside Qarmatians
			}
		}
	}
	
	major_trigger = {
		religion_group = muslim
		has_discovered = 408
		NOT = {
			tag = QRM		#Qarmatians have a separate event
		}
	}
	
	immediate = {
		hidden_effect = {
			QRM = {
				country_event = {		#triggers notification event for Qarmatians
					id = mmetrFlavor_War.2
				}
			}
			QRM = {
				add_prestige = 5
				add_treasury = 50
				capital_scope = {
					add_base_production = 1
				}
			}
			if = {
				limit = {
					base_tax = 2
				}
				add_base_tax = -1
			}
			if = {
				limit = {
					base_manpower = 2
				}
				add_base_manpower = -1
			}
		}
	}
	
	fire_only_once = yes
	
	option = {
		name = "mmetrFlavor_War.1.a"
		
		custom_tooltip = mmetrFlavor_War.1.tt
		tooltip = {
			QRM = {
				add_prestige = 5
				add_treasury = 50
				capital_scope = {
					add_base_production = 1
				}
			}
			if = {
				limit = {
					base_tax = 2
				}
				add_base_tax = -1
			}
			if = {
				limit = {
					base_manpower = 2
				}
				add_base_manpower = -1
			}
		}
	}
}

# Province Defects to Palmyra - Rome POV
province_event = {
	id = mmetrFlavor_War.3
	title = "mmetrFlavor_War.3.t"
	desc = "mmetrFlavor_War.3.desc"
	picture = SIEGE_eventPicture
	
	trigger = {
		controller = {
			is_in_war = {
				casus_belli = cb_claim_eastern_provinces
				defenders = ROM
			}
		}
		owner = {
			OR = {
				tag = ROM
				tag = BYZ
				tag = ROW
			}
		}
		OR = {
			region = egypt_region
			region = anatolia_region
			region = mashriq_region
		}
	}
	mean_time_to_happen = {
		months = 60
		modifier = {
			factor = 0.5
			any_neighbor_province = {
				owner = {
					is_in_war = {
						casus_belli = cb_claim_eastern_provinces
					}
				}
			}
		}
		modifier = {
			factor = 0.5
			all_neighbor_province = {
				controller = { 
					is_in_war = {
						casus_belli = cb_claim_eastern_provinces
					}
				}
			}
		}
	}
	
	immediate = {
		hidden_effect = {
			controller = {
				country_event = {
					id = mmetrFlavor_War.4
				}
			}
		}
	}
	
	option = {
		name = "mmetrFlavor_War.3.a"
		
		owner = { add_prestige = -5 }
		custom_tooltip = mmetrFlavor_War.3.tt
	}
}

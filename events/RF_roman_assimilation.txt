#Iberia
country_event = {
    id = roman_assimilation.1
    title = roman_assimilation.1.t
    desc = roman_assimilation.1.d
    picture = CULTURE_eventPicture

    trigger = {
		is_at_war = no
        OR = {
            tag = ROM
            primary_culture = roman
        }
        any_owned_province = {
            region = iberia_region
            NOT = { culture = hispanian }
            has_owner_culture = no
			has_owner_religion = yes
            NOT = {unrest = 0}
            NOT = {nationalism = 1}
        }
    }

    mean_time_to_happen = {
        months = 360
        modifier = {
            factor = 0.5
            has_country_modifier = pax_romana
        }
        modifier = {
            factor = 0.8
            stability = 3
        }
        modifier = {
            factor = 1.25
            NOT = {stability = 1}
        }
    }


    option = {
        name = roman_assimilation.1.a
        ai_chance = { factor = 90 }

        random_owned_province = {
            limit = {
				region = iberia_region
				NOT = { culture = hispanian }
				has_owner_culture = no
				has_owner_religion = yes
				NOT = {unrest = 0}
				NOT = {nationalism = 1}
            }
            change_culture = hispanian
            add_province_modifier = {
                name = romanized_identity
                duration = 7300 # ~20 years flavor
            }
        }
    }

    option = {
        name = roman_assimilation.1.b
        ai_chance = { factor = 10 }
        add_prestige = 2
    }
}

# The Walls of King Herod
country_event = {
	id = mmetr_JUD_mission_events.1
	title = "mmetr_JUD_mission_events.1.t"
	desc = "mmetr_JUD_mission_events.1.desc"
	picture = FORT_eventPicture

	is_triggered_only = yes
	
	option = {
		name = "mmetr_JUD_mission_events.1.a"
		
		379 = {
			add_province_modifier = {
				name = "me_jud_herodian_walls"
				duration = -1
			}
		}
		if = {
			limit = {
				has_estate = estate_church
			}
			add_estate_loyalty = {
				estate = estate_church
				loyalty = 10
			}
		}
		else = {
			add_prestige = 10
			add_church_power = 10
		}
	}
	
	option = {
		name = "mmetr_JUD_mission_events.1.b"
		
		379 = {
			add_province_modifier = {
				name = "me_jud_herodian_walls"
				duration = -1
			}
		}
		add_manpower = 5
	}
}

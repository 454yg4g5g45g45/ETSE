# --------------------------------------------------
# EVENT 1: POSITIVE (Powerful) - Dies Natalis Solis Invicti
# Context: The Birthday of the Unconquered Sun (Dec 25).
# --------------------------------------------------
country_event = {
    id = sol_flavor.1
    title = "sol_flavor.1.t"
    desc = "sol_flavor.1.d"
    picture = RELIGION_eventPicture
    
    trigger = {
        religion = sol_invictus
        NOT = { has_country_modifier = sol_invictus_triumph }
    }
    
    mean_time_to_happen = { months = 480 }
    
    option = {
        name = "sol_flavor.1.a" # The Sun is reborn!
        add_stability = 1
        add_prestige = 50
        add_mil_power = 100 # Military focus
		add_republican_tradition = 10
		add_legitimacy = 15
		add_devotion = 15
        
        add_country_modifier = {
            name = sol_invictus_triumph
            duration = 7300 # 20 Years
        }
    }
}

# --------------------------------------------------
# EVENT 6: NEGATIVE (Zealotry) - Iconoclasm
# Context: Zealots destroying "lesser" gods' temples.
# --------------------------------------------------
country_event = {
    id = sol_flavor.6
    title = "sol_flavor.6.t"
    desc = "sol_flavor.6.d"
    picture = RELIGIOUS_TURMOIL_eventPicture
    
    trigger = {
        religion = sol_invictus
        any_owned_province = {
            NOT = { religion = sol_invictus }
        }
    }
    
    mean_time_to_happen = { months = 500 }
    
    option = {
        name = "sol_flavor.6.a" # Let them burn the idols!
        add_prestige = -10
        add_local_autonomy = 10 # Disorder
        # Spawns destruction in a random heathen province
        random_owned_province = {
            limit = { NOT = { religion = sol_invictus } }
            add_devastation = 30
            change_religion = sol_invictus # But it converts!
        }
    }
    
    option = {
        name = "sol_flavor.6.b" # Order must be maintained.
        add_adm_power = -25
        random_owned_province = {
            limit = { NOT = { religion = sol_invictus } }
            spawn_rebels = {
                type = religious_rebels # Sol Invictus zealots
                size = 1
            }
        }
    }
}

# The Faith of the Empire
country_event = {
	id = mmetr_ROM_mission_events.7
	title = "mmetr_ROM_mission_events.7.t"
	desc = "mmetr_ROM_mission_events.7.desc"
	picture = CPBM_ROMAN_TEMPLE_eventPicture

	is_triggered_only = yes

	option = {
		name = "mmetr_ROM_mission_events.7.a"	#if is neither Christian nor Hellenic - if player decides to go other religion
		
		highlight = yes
		trigger = {
			NOT = {
				religion_group = christian
				religion = hellenism
			}
		}
		
		add_country_modifier = {
			name = "me_rom_true_piety"
			duration = 10950
		}
		swap_non_generic_missions = yes
		
		ai_chance = {		#ai will not convert
			factor = 1
			modifier = {
				factor = 0
				OR = {
					religion_group = christian
					religion = hellenism
				}
			}
		}
	}
	
	option = {
		name = "mmetr_ROM_mission_events.7.b"	#Christian path, if started BEFORE Rome converted to Christianity
		
		trigger = {
			NOT = { started_in = 2478.2.27 }
		}
		
		if = {
			NOT =  {
				religion_group = christian
			}
			change_religion = chalcedonism
			custom_tooltip = mmetr_ROM_mission_events.7.tt		#tooltip for new branching missions
		}
		add_country_modifier = {
			name = "me_rom_christian_mission"
			duration = 10950
		}
		swap_non_generic_missions = yes
		
		ai_chance = {		#ai will not convert
			factor = 1
			modifier = {
				factor = 0
				NOT = { religion_group = christian }
			}
		}
	}
	
	option = {
		name = "mmetr_ROM_mission_events.7.c"	#Christian path, if AFTER Roman conversion to Christianity
		
		trigger = {
			started_in = 2478.2.27
		}
		
		if = {
			NOT =  {
				religion_group = christian
			}
			change_religion = chalcedonism
			custom_tooltip = mmetr_ROM_mission_events.7.tt		#tooltip for new branching missions
		}
		add_country_modifier = {
			name = "me_rom_christianity_affirmed"
			duration = 10950
		}
		swap_non_generic_missions = yes
		
		ai_chance = {		#ai will not convert
			factor = 1
			modifier = {
				factor = 0
				religion_group = christian
			}
		}
	}
	
	option = {
		name = "mmetr_ROM_mission_events.7.d"	#Hellenic path, started AFTER Rome converted to Christianity - ex: Julian the Apostate
		
		trigger = {
			started_in = 2478.2.27
		}
		
		if = {
			NOT =  {
				religion = hellenism
			}
			change_religion = hellenism
			custom_tooltip = mmetr_ROM_mission_events.7.tt		#tooltip for new branching missions
		}
		add_country_modifier = {
			name = "me_rom_pagan_survival"
			duration = 10950
		}
		swap_non_generic_missions = yes
		
		ai_chance = {		#ai will not convert
			factor = 1
			modifier = {
				factor = 0
				religion_group = christian
			}
		}
	}

	option = {
		name = "mmetr_ROM_mission_events.7.e"	#Hellenic path, if BEFORE Roman conversion to Christianity
		
		trigger = {
			NOT = { started_in = 2478.2.27 }
		}
		
		if = {
			NOT =  {
				religion = hellenism
			}
			change_religion = hellenism
			custom_tooltip = mmetr_SLL_mission_events.1.tt1		#tooltip for new branching missions
		}
		add_country_modifier = {
			name = "me_rom_imposed_imperial_pantheon"
			duration = 10950
		}
		swap_non_generic_missions = yes
		
		ai_chance = {		#ai will not convert
			factor = 1
			modifier = {
				factor = 0
				religion_group = christian
			}
		}
	}
}

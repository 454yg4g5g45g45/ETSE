			limit = {

			}

			name = "me_genchnw_legalist_policies"
			duration = 9125
		}

			change_religion = daoism
		}
		add_country_modifier = {
			name = "me_genchnw_daoist_masters"

		}
	}

		}
		change_religion = mahayana
		capital_scope = {
			change_religion = mahayana
		}
		add_country_modifier = {
			name = "me_genchnw_buddhist_doctrines"
			duration = 9125
		}
	}
	
	option = {

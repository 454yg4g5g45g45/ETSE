	title = "roman_characters_event.1.t"

	trigger = {
		tag = ROM
		is_year = 2248
		NOT = { is_year = 2314 }
		owns_core_province = 325 # Castamonu
		NOT = { has_global_flag = galen_recruited }
	}
	mean_time_to_happen = { months = 12 }
	immediate = { 325 = { save_event_target_as = char_origin_province } }
	
	option = {
		name = "roman_characters_event.1.opt1" # Advisor (Natural Scientist)
		set_global_flag = galen_recruited
		add_years_of_income = -0.25
		define_advisor = {
			name = "Claudius Galenus"
			type = natural_scientist
			skill = 3
			discount = yes
			culture = event_target:char_origin_province

	}
	option = {
		name = "roman_characters_event.1.refuse"
		ai_chance = { factor = 10 }

	}
}

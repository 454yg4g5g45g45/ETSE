custom_button = {
    name = sacrifice_messenger_button
    potential = {
        religion = zamolxism
    }
    trigger = {
		NOT = { has_country_modifier = sacrificed_messenger_wealth }
		NOT = { has_country_modifier = sacrificed_messenger_abundance }
		NOT = { has_country_modifier = sacrificed_messenger_might }
	}
    effect = {
		if = {
			limit = {
				has_country_flag = open_sacrifice_messenger_window_flag
			}
			clr_country_flag = open_sacrifice_messenger_window_flag
		}
		else = {
			set_country_flag = open_sacrifice_messenger_window_flag
			
			play_sound = start_convert_shamanism	 #new in 1.36
		}
	}
    tooltip = sacrifice_messenger_button.tt
}

custom_button = {
    name = hunnic_zun_button
    potential = {
        religion = hunnic_religion
		has_country_flag = open_hunnic_faith_actions_window_flag
    }
    trigger = {
		dip_power = 100		#always available
		custom_trigger_tooltip = {
			tooltip = hunnic_religion_max_syncretic_aspects
			NOT = {
				calc_true_if = {
					has_country_modifier = hunnic_zun_button_modifier
					has_country_modifier = hunnic_buddhist_button_modifier
					has_country_modifier = hunnic_hindu_button_modifier
					has_country_modifier = hunnic_jain_button_modifier
					has_country_modifier = hunnic_fire_temple_button_modifier
					has_country_modifier = hunnic_tengri_button_modifier
					has_country_modifier = hunnic_tabiti_button_modifier
					has_country_modifier = hunnic_nestorian_button_modifier
					has_country_modifier = hunnic_muslim_merchant_button_modifier
					has_country_modifier = hunnic_eastern_schools_button_modifier
					amount = 3
				}
			}
		}
		NOT = {
			has_country_modifier = hunnic_zun_button_modifier
		}
	}
    effect = {
		add_dip_power = -100
		add_country_modifier = {
			name = "hunnic_zun_button_modifier"
			duration = 5475
		}
		custom_tooltip = hunnic_zun_button.tt2
		hidden_effect = {
			every_owned_province = {
				limit = {
					OR = {
						religion = zunism
						religion = mithraic
						religion = khurmazta
					}
				}
				add_province_modifier = {
					name = "hunnic_promoted_local_religion"
					duration = 5475
				}
			}
		}
		clr_country_flag = open_hunnic_faith_actions_window_flag
		play_sound = start_convert_shamanism	 #sound SFX, new in 1.36
	}
    tooltip = hunnic_zun_button.tt
}

cb_raid_pillage = {
	valid_for_subject = no
	prerequisites_self = {
		technology_group = nomad_group
	}
	prerequisites = {
		FROM = {
			OR = {
				is_neighbor_of = ROOT
				any_neighbor_country = {
					OR = {
						is_subject_of = ROOT
						root = {
							has_casus_belli = {
								type = cb_raid_pillage
								target = THIS
							}
						}
						gives_military_access_to = ROOT
					}
				}
			}
		}
	}
	war_goal = wg_raid_pillage
}

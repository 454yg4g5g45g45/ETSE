estate_scholars_disaster = {
	potential = {
		has_estate = estate_scholars
		estate_influence = {
			estate = estate_scholars
			influence = 30
		}
	}


	can_start = {
		has_any_disaster = no
		is_at_war = no
		OR = {
			estate_influence = {
				estate = estate_scholars
				influence = 100
			}
			AND = {
				has_regency = yes
				estate_influence = {
					estate = estate_scholars
					influence = 90
				}
			}
		}
	}
	
	can_stop = {
		OR = {
			has_any_disaster = yes
			NOT = {
				estate_influence = {
					estate = estate_scholars
					influence = 100
				}		
			}
		}
	}
	
	down_progress = {
		factor = 0
		modifier = {
			factor = 1
			NOT = {
				estate_influence = {
					estate = estate_scholars
					influence = 100
				}
			}
		}
	}
	
	progress = {
		modifier = {
			factor = 0.5
			NOT = {
				estate_loyalty = {
					estate = estate_scholars
					loyalty = 50
				}
			}
			is_at_war = no
		}
		modifier = {
			factor = 0.5
			NOT = {
				estate_loyalty = {
					estate = estate_scholars
					loyalty = 40
				}
			}
			is_at_war = no
		}
		modifier = {
			factor = 1
			NOT = {
				estate_loyalty = {
					estate = estate_scholars
					loyalty = 30
				}
			}
			is_at_war = no
		}
		modifier = {
			factor = 1
			NOT = {
				estate_loyalty = {
					estate = estate_scholars
					loyalty = 20
				}
			}
			is_at_war = no
		}
	}
	
	can_end = {
		custom_trigger_tooltip = {
			tooltip = EST_CRUSHED_EUNUCHS
			NOT = {
				has_country_flag = eunuchs_estate_in_power
			}
		}
	}
	
	modifier = {
		global_unrest = 1
		administrative_efficiency = -0.05
		yearly_corruption = 0.1
	}

	on_start = estate_disasters_mmetr.1
	on_end = estate_disasters_mmetr.2
	
	on_monthly = {
		
	}
}

uprising_of_five_hu = {
	potential = {
		NOT = { has_country_flag = uprising_of_five_hu_happened }
		normal_or_historical_nations = yes
		is_year = 2378
		NOT = { started_in = 2414.1.1 }
		OR = {
			is_emperor_of_china = yes
			government_rank = 10
			tag = JIN
			was_tag = JIN
		}
	}
	
	can_start = {
		has_any_disaster = no
		num_of_cities = 60
		num_of_owned_provinces_with = {
			value = 20
			region = north_china_region
		}
		OR = {
			is_emperor_of_china = yes
			government_rank = 10
		}
		OR = {
			NOT = { stability = 0 }
			estate_influence = {
				estate = estate_nobles
				influence = 75
			}
			NOT = { adm = 2 }
			NOT = { mil = 2 }
			has_regency = yes
		}
		if = {
			limit = {	
				has_dlc = "Mandate of Heaven"
			}
			OR = {
				AND = {
					NOT = { imperial_mandate = 2148 }
					is_emperor_of_china = yes
				}
				NOT = { stability = -1 }		#changed from -2 to -1
				has_regency = yes
				is_bankrupt = yes
				
				#only for AI
				hidden_trigger = {
					ai = yes
					NOT = { stability = 2 }		#triggers more frequently for AI
				}
			}
		}
		else = {
			OR = {
				has_country_modifier = the_mandate_of_heaven_lost
				has_country_modifier = lost_mandate_of_heaven
				NOT = { legitimacy = 50 }
				NOT = { stability = -1 }		#changed from -2 to -1
				has_regency = yes
				is_bankrupt = yes
				
				#only for AI
				hidden_trigger = {
					ai = yes
					NOT = { stability = 2 }		#triggers more frequently for AI
				}
			}
		}
	}
	
	can_stop = {
		OR = {
			has_any_disaster = yes
			stability = 3
		}
	}
	
	progress = {
		modifier = {
			factor = 10
			has_dlc = "Mandate of Heaven"
			NOT = { is_emperor_of_china = yes }
		}
		modifier = {
			factor = 2
			NOT = { imperial_mandate = 2148 }
			imperial_mandate = 2128
		}
		modifier = {
			factor = 3
			NOT = { imperial_mandate = 2128 }
			imperial_mandate = 2108
		}
		modifier = {
			factor = 10
			NOT = { imperial_mandate = 2108 }
		}
		modifier = {
			factor = 1
			num_of_loans = 5
		}
		modifier = {
			factor = 3
			num_of_loans = 10
		}
		modifier = {
			factor = 5
			is_bankrupt = yes
		}
		modifier = {
			factor = 2
			corruption = 5
			NOT = { corruption = 10 }
		}
		modifier = {
			factor = 5
			corruption = 10
		}
		modifier = {
			factor = 2
			NOT = { stability = 0 }
		}
		modifier = {
			factor = 5
			NOT = { stability = -1 }
		}
		modifier = {
			factor = 10
			NOT = { stability = -2 }
		}
	}
	
	can_end = {
		if = {
			limit = {
				has_dlc = "Mandate of Heaven"
			}
			OR = {
				imperial_mandate = 2168
				is_emperor_of_china = no
			}
		}
		else = {
			stability = 3
		}
		stability = 2
		is_bankrupt = no
		NOT = { num_of_loans = 5 }
		NOT = { corruption = 5 }
		NOT = { num_of_rebel_armies = 1 }
		NOT = { num_of_rebel_controlled_provinces = 1 }
	}
	
	modifier = {
		land_morale = -0.15
		technology_cost = 0.1
		global_unrest = 12
		global_tax_modifier = -0.25
		global_trade_goods_size_modifier = -0.2
	}
	
	on_start = uprising_of_five_hu.1
	on_end = uprising_of_five_hu.2
	
	on_monthly = {
		events = {
			uprising_of_five_hu.3	#Qi Wannian Rebels
			uprising_of_five_hu.4	#Li Te Rebels in Bashu
			uprising_of_five_hu.5	#Li Clan Establishes Cheng-Han
			uprising_of_five_hu.11	#Liu Yuan Conquers Luoyang / Disaster of Yongjia
		}
		random_events = {
			2500 = 0
			100 = uprising_of_five_hu.6		#Rebellions in Hebei
			100 = uprising_of_five_hu.7		#Rebel Remnants Strike Qing and Xu
			100 = uprising_of_five_hu.8		#Chen Min Revolts in Jiangnan
			100 = uprising_of_five_hu.9		#Zhang Chang Rebels in Jingzhou
			300 = uprising_of_five_hu.10	#Liu Yuan Invades the North
		}
	}
}
